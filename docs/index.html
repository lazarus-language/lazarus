Variables & affichage

laz nom = "Ladji"
laz age = 25
age = age + 1
vox("Salut", nom)

Conditions

kan age >= 18 {
    vox("Majeur")
} sinon kan age >= 13 {
    vox("Ado")
} sinon {
    vox("Enfant")
}
# logique : et / ou / non

Boucles

tanke x > 0 {
    x = x - 1
}
pou i dan 1..10 {
    vox(i)
}
# kase = break, swiv = continue

Fonctions

fonk carre(x) {
    rend x * x
}
vox(carre(8))

Listes

laz l = [1, 2, 3]
vox(l[0], taille(l))
ajoute(l, 4)
l[1] = 99
vox(tri(l))

Classes & objets (v2.0)

klas Chien herite Animal {
    fonk init(moi, nom) {
        moi.nom = nom
    }
    fonk parler(moi) {
        vox(moi.nom, "dit wouf")
    }
}
laz rex = Chien("Rex")
rex.parler()

Dictionnaires (v2.0)
